#!/usr/bin/env python3 """ Personal Command-Line AI Assistant for Termux & PC Author: Devesh Project: DTA """ import os import sys import webbrowser import platform import subprocess def clear_screen(): os.system('cls' if platform.system() == 'Windows' else 'clear') def speak_or_print(text): print(f"\n[AI Assistant]: {text}\n") def execute_command(user_input): cmd = user_input.lower().strip() # 1. Instagram ओपन करने की कमांड if "instagram" in cmd or "इंस्टाग्राम" in cmd: speak_or_print("Instagram खोला जा रहा है...") if platform.system() == "Linux" and os.path.exists("/data/data/com.termux"): # Termux के लिए Android Intent का इस्तेमाल try: subprocess.run([ "am", "start", "-a", "android.intent.action.VIEW", "-d", "https://www.instagram.com" ]) except Exception: webbrowser.open("https://www.instagram.com") else: # PC या अन्य डिवाइस के लिए Browser webbrowser.open("https://www.instagram.com") return True # 2. YouTube चलाने की कमांड elif "youtube" in cmd or "यूट्यूब" in cmd: speak_or_print("YouTube चलाया जा रहा है...") if platform.system() == "Linux" and os.path.exists("/data/data/com.termux"): try: subprocess.run([ "am", "start", "-a", "android.intent.action.VIEW", "-d", "https://www.youtube.com" ]) except Exception: webbrowser.open("https://www.youtube.com") else: webbrowser.open("https://www.youtube.com") return True # 3. Google पर सर्च करने की कमांड (जैसे "Google पर X सर्च कर") elif "google पर" in cmd or "search on google" in cmd or "सर्च कर" in cmd: # कीवर्ड निकालने की कोशिश for phrase in ["google पर", "search on google", "सर्च कर", "search"]: if phrase in cmd: query = cmd.split(phrase)[-1].strip() break else: query = cmd speak_or_print(f"Google पर '{query}' खोजा जा रहा है...") search_url = f"https://www.google.com/search?q={query}" webbrowser.open(search_url) return True # 4. सिस्टम ऐप्स या सामान्य कमांड (बाहर निकलना) elif cmd in ["exit", "quit", "बाहर जाओ", "बंद करो"]: speak_or_print("अलविदा! अपना ध्यान रखें।") sys.exit(0) # 5. सामान्य AI रिस्पॉन्स (यहाँ आप Gemini API या अन्य AI इंटीग्रेट कर सकते हैं) else: return False def main(): clear_screen() print("=" * 50) print(" DTA Personal Command-Line AI Assistant ") print("=" * 50) speak_or_print("नमस्ते! मैं आपका पर्सनल असिस्टेंट हूँ। बताइए मैं आपकी क्या मदद कर सकता हूँ?") while True: try: user_input = input("आप (You) > ") if not user_input.strip(): continue # कमांड चेक करें और एक्सेक्यूट करें handled = execute_command(user_input) if not handled: # यदि कमांड सीधे मैच न हो, तो यहाँ AI लॉजिक काम करेगा speak_or_print(f"माफ कीजिए, मुझे '{user_input}' के लिए कोई डायरेक्ट कमांड नहीं मिली। आप इसे किसी AI API (जैसे Gemini) से कनेक्ट कर सकते हैं।") except KeyboardInterrupt: print("\n") speak_or_print("प्रोग्राम बंद किया जा रहा है। अलविदा!") break if __name__ == "__main__": main()
